const app = document.getElementById('app');

const parse = s => { const [y,m,d] = s.split('-').map(Number); return new Date(y, m-1, d); };

const fmt = (d,o) => d.toLocaleDateString('en-GB', o);

const longDate = s => { const d = parse(s); return fmt(d,{weekday:'short'}) + ' ' + d.getDate() + ' ' + fmt(d,{month:'long'}); };

const desc = (a,b) => b.date.localeCompare(a.date);

const asc = (a,b) => a.date.localeCompare(b.date);

const taught = () => LESSONS.filter(l => !l.cancelled);


function h(tag, props = {}, ...kids) {

  const e = document.createElement(tag);

  for (const [k,v] of Object.entries(props)) {

    if (k === 'class') e.className = v;

    else if (k === 'text') e.textContent = v;

    else e.setAttribute(k, v);

  }

  kids.flat().forEach(c => { if (c == null) return; e.append(c.nodeType ? c : document.createTextNode(c)); });

  return e;

}

const safe = u => /^https?:\/\//i.test(u || '');

function list(arr, empty) {

  if (!arr || !arr.length) return h('p', {class:'soon', text: empty || 'Coming soon'});

  return h('ul', {}, arr.map(t => h('li', {text: t})));

}

function links(arr) {

  const ok = (arr || []).filter(x => safe(x.url));

  if (!ok.length) return h('p', {class:'soon', text:'Coming soon'});

  return h('ul', {class:'plain'}, ok.map(x => h('li', {}, h('a', {class:'btn small', href:x.url, target:'_blank', rel:'noopener', text:x.label}))));

}

function chatLinks() {

  const items = [];

  if (safe(CONFIG.padletQA)) items.push({label:'Ask the teacher (Q&A)', url:CONFIG.padletQA});

  if (safe(CONFIG.padletChat)) items.push({label:'Class chat', url:CONFIG.padletChat});

  return links(items);

}

const part = (title, ...content) => h('div', {class:'part'}, h('h3', {text:title}), content);

const activeBooks = () => [...new Set(LESSONS.map(l => l.book))].sort((a,b) => a-b);


function lessonBlock(l) {

  const d = parse(l.date);

  const badge = h('div', {class:'badge'},

    h('span', {class:'wd', text: fmt(d,{weekday:'short'}).toUpperCase()}),

    h('span', {class:'day', text: String(d.getDate())}),

    h('span', {class:'mo', text: fmt(d,{month:'long'}).toUpperCase()}));

  const box = h('article', {class:'card lesson', id:'lesson-' + l.date}, badge);

  if (l.cancelled) {

    box.append(h('div', {class:'parts'}, h('div', {class:'part wide'}, h('h3', {text:'No lesson'}), h('p', {text:l.note || 'This lesson was cancelled.'}))));

    return box;

  }

  box.append(h('div', {class:'parts'},

    part('What we did', list(l.did), l.test ? h('p', {class:'test', text:'Short test: ' + l.test}) : null),

    part('Homework', list(l.homework, 'No homework')),

    part('Practice and extra resources', links(l.practice)),

    part('Games and fun practice', links(l.games)),

    part('Ask / chat', chatLinks())));

  return box;

}


function homePage() {

  const wrap = h('div');

  const rules = h('details', {class:'card'}, h('summary', {class:'btn', text:'Class rules'}), h('div', {class:'inner'}, list(CONFIG.rules)));

  const chat = h('div', {class:'card', style:'padding:12px'}, h('div', {}, chatLinks()));

  chat.prepend(h('h3', {text:'Ask and chat'}));

  wrap.append(h('div', {class:'strip'}, rules, dates, chat));


  const sorted = [...LESSONS].sort(desc);

  const latest = sorted.find(l => !l.cancelled);

  if (latest) {

    wrap.append(h('section', {class:'card latest'},

      h('h3', {text:'Latest lesson'}),

      h('p', {class:'big-date', text: longDate(latest.date)}),

      part('What we did', list(latest.did), latest.test ? h('p', {class:'test', text:'Short test: ' + latest.test}) : null),

      part('Homework', list(latest.homework, 'No homework')),

      h('p', {}, h('a', {class:'btn', href:'#book-' + latest.book + '/' + latest.date, text:'Open full lesson'}))));

    wrap.append(h('h2', {text:'Previous lessons'}));

    sorted.filter(l => l !== latest).slice(0, 4).forEach(l => {

      wrap.append(h('a', {class:'row', href:'#book-' + l.book + '/' + l.date},

        h('b', {text: longDate(l.date)}),

        h('span', {text: l.cancelled ? 'No lesson' : (l.did || []).join(', ')})));

    });

  }

  wrap.append(h('h2', {text:'Books'}));

  const act = activeBooks();

  const grid = h('div', {class:'grid'});

  CONFIG.books.forEach(n => {

    grid.append(act.includes(n)

      : h('span', {class:'btn off', text:'Book ' + n + ' (soon)'}));

  });

  wrap.append(grid);

  wrap.append(h('p', {}, h('a', {class:'btn', href:'#revision', text:'Revision and exams'}), ' ', h('a', {class:'btn', href:'#ket', text:'KET (A2 Key) corner'})));

function bookPage(n) {

  const wrap = h('div', {}, h('p', {}, h('a', {href:'#home', text:'← Back to home'})), h('h2', {text:'Book ' + n + ' – Lessons'}));

  const ls = LESSONS.filter(l => l.book === n).sort(desc);

  if (!ls.length) wrap.append(h('p', {class:'card soon', text:'Coming soon.'}));

  ls.forEach(l => wrap.append(lessonBlock(l)));

  return wrap;

}


function store() { try { return JSON.parse(localStorage.getItem('hub-checks') || '{}'); } catch (e) { return {}; } }

function saveStore(o) { try { localStorage.setItem('hub-checks', JSON.stringify(o)); } catch (e) {} }


function checklist(key, filterFn) {

  const seen = new Set(), items = [];

  taught().filter(filterFn).sort(asc).forEach(l => (l.did || []).forEach(t => { if (!seen.has(t)) { seen.add(t); items.push(t); } }));

  const saved = store();

  const box = h('div', {});

  if (!items.length) box.append(h('p', {class:'soon', text:'Topics appear here after each lesson.'}));

  items.forEach(t => {

    const id = key + '|' + t;

    const cb = h('input', {type:'checkbox'});

    cb.checked = !!saved[id];

    cb.addEventListener('change', () => { const s = store(); s[id] = cb.checked; saveStore(s); });

    box.append(h('label', {class:'chk'}, cb, h('span', {text:t})));

  });

  return box;

}


function revisionPage() {

  const wrap = h('div', {}, h('p', {}, h('a', {href:'#home', text:'← Back to home'})), h('h2', {text:'Revision and exams'}),

    h('p', {text:'Topics are added automatically after each lesson. Tick what you have revised.'}));

  const section = (title, key, fn, open) => {

    const d = h('details', {class:'card'}, h('summary', {class:'btn', text:title}),

      h('div', {class:'inner'}, checklist(key, fn), h('h3', {text:'Extra practice'}), links(CONFIG.revisionLinks[key])));

    if (open) d.setAttribute('open', '');

    return d;

  };

  activeBooks().forEach(n => wrap.append(section('Book ' + n + ' test revision', 'book-' + n, l => l.book === n, true)));

  wrap.append(section('Semester exam revision', 'semester', l => l.date <= CONFIG.semesterEnd));

  wrap.append(section('End of year exam revision', 'year', () => true));

  return wrap;

}


function ketPage() {

  const k = CONFIG.ket;

  const block = (title, content) => h('section', {class:'card'}, h('h3', {text:title}), content);

  return h('div', {}, h('p', {}, h('a', {href:'#home', text:'← Back to home'})), h('h2', {text:'KET (A2 Key) corner'}),

    block('What is in the exam', list(k.about)),

    block('Official free practice', links(k.official)),

    block('Our vocabulary Quizlets', links(k.quizlets)),

    block('Mock exam dates', list((k.mocks || []).map(m => m.label + (m.note ? ': ' + m.note : '')), 'Coming soon')));

}


function render() {

  const [page, anchor] = (location.hash.slice(1) || 'home').split('/');

  app.replaceChildren();

  if (page.startsWith('book-')) app.append(bookPage(Number(page.slice(5))));

  else if (page === 'revision') app.append(revisionPage());

  else if (page === 'ket') app.append(ketPage());

  else app.append(homePage());

  const cur = page.startsWith('book-') ? 'home' : page;

  document.querySelectorAll('nav a').forEach(a => a.classList.toggle('on', a.getAttribute('href') === '#' + cur));

  const target = anchor && document.getElementById('lesson-' + anchor);

  if (target) target.scrollIntoView(); else window.scrollTo(0, 0);

}

window.addEventListener('hashchange', render);

render();
